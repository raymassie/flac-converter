# Audio Converter dependencies
#
# System requirements (not installable via pip):
#   - Python 3.9 or higher, built with tkinter support
#     (Homebrew: brew install python-tk)
#   - ffmpeg          (Homebrew: brew install ffmpeg)

# Drag-and-drop support for files and folders.
# Without it the app still runs; the Select button is the only way to add files.
tkinterdnd2>=0.3.0

# Cross-platform "move to Trash" for the Delete original files option.
# On macOS the app falls back to Finder via osascript if this is absent.
send2trash>=1.8.0
